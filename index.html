var ss = SpreadsheetApp.getActiveSpreadsheet(); function doGet(e) { var action = e.parameter.action; if (action === 'admin_data') { return getAdminData(); } else if (action === 'absen') { var nis = e.parameter.nis; var manualStatus = e.parameter.status; return prosesAbsen(nis, manualStatus); } else if (action === 'tambah_kelas') { return tambahKelas(e.parameter.kelas); } else if (action === 'tambah_guru') { return tambahGuru(e.parameter); } return ContentService.createTextOutput(JSON.stringify({status: "error", message: "Invalid action"})).setMimeType(ContentService.MimeType.JSON); } // Ambil batas jam dari sheet Config function getBatasJamConfig() { var sheetConfig = ss.getSheetByName("Config"); var batasJam = "07:30"; // Default jika sheet config kosong if (sheetConfig) { var data = sheetConfig.getDataRange().getValues(); for (var i = 0; i < data.length; i++) { if (data[i][0] && data[i][0].toString().toLowerCase().indexOf("batas") !== -1 || data[i][0].toString().indexOf("07:") !== -1) { batasJam = data[i][1] ? data[i][1].toString() : data[i][0].toString(); break; } } } return batasJam; } function getAdminData() { var sheetSiswa = ss.getSheetByName("Siswa"); var sheetAbsen = ss.getSheetByName("Absensi"); var listSiswa = []; if (sheetSiswa) { var dataSiswa = sheetSiswa.getDataRange().getValues(); for (var i = 1; i < dataSiswa.length; i++) { if (dataSiswa[i][0]) { listSiswa.push({ nis: dataSiswa[i][0].toString(), nama: dataSiswa[i][1], kelas: dataSiswa[i][2], wali: dataSiswa[i][3] || "-" }); } } } var listHadir = []; var listTelat = []; if (sheetAbsen) { var dataAbsen = sheetAbsen.getDataRange().getValues(); var todayStr = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd"); for (var j = 1; j < dataAbsen.length; j++) { var row = dataAbsen[j]; if (row[1]) { var item = { nis: row[1].toString(), nama: row[2], kelas: row[3], waktu: row[5], status: row[6] }; // Klasifikasikan berdasarkan kolom status di spreadsheet (Hadir / Telat / Izin / Sakit) var stat = (row[6] ? row[6].toString().trim() : "Hadir"); if (stat === 'Telat') { listTelat.push(item); } else { listHadir.push(item); } } } } var result = { totalSiswa: listSiswa.length, totalHadir: listHadir.length, totalTelat: listTelat.length, listSiswa: listSiswa, listHadir: listHadir, listTelat: listTelat }; return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON); } function prosesAbsen(nis, manualStatus) { var sheetSiswa = ss.getSheetByName("Siswa"); var sheetAbsen = ss.getSheetByName("Absensi"); if (!sheetSiswa || !sheetAbsen) { return ContentService.createTextOutput(JSON.stringify({status: "error", message: "Sheet tidak ditemukan"})).setMimeType(ContentService.MimeType.JSON); } var dataSiswa = sheetSiswa.getDataRange().getValues(); var siswaDitemukan = null; for (var i = 1; i < dataSiswa.length; i++) { if (dataSiswa[i][0].toString() === nis.toString()) { siswaDitemukan = { nis: dataSiswa[i][0].toString(), nama: dataSiswa[i][1], kelas: dataSiswa[i][2] }; break; } } if (!siswaDitemukan) { return ContentService.createTextOutput(JSON.stringify({status: "error", message: "Siswa dengan NIS " + nis + " tidak terdaftar!"})).setMimeType(ContentService.MimeType.JSON); } var now = new Date(); var tanggalStr = Utilities.formatDate(now, Session.getScriptTimeZone(), "yyyy-MM-dd"); var jamStr = Utilities.formatDate(now, Session.getScriptTimeZone(), "HH:mm"); var timestamp = Utilities.formatDate(now, Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm:ss"); var statusKehadiran = "Hadir"; if (manualStatus) { statusKehadiran = manualStatus; if (manualStatus === "Belum Absen") { var dataAbsen = sheetAbsen.getDataRange().getValues(); for (var r = dataAbsen.length - 1; r >= 1; r--) { if (dataAbsen[r][1].toString() === nis.toString()) { sheetAbsen.deleteRow(r + 1); } } return ContentService.createTextOutput(JSON.stringify({status: "success", message: "Absen direset"})).setMimeType(ContentService.MimeType.JSON); } } else { // Ambil batas jam dari Config atau default 07:30 var batasConfig = getBatasJamConfig(); // contoh "07:30" var partsBatas = batasConfig.split(":"); var limitMenit = parseInt(partsBatas[0] || 7) * 60 + parseInt(partsBatas[1] || 30); var jamParts = jamStr.split(":"); var totalMenitSekarang = parseInt(jamParts[0]) * 60 + parseInt(jamParts[1]); if (totalMenitSekarang > limitMenit) { statusKehadiran = "Telat"; } else { statusKehadiran = "Hadir"; } } sheetAbsen.appendRow([timestamp, siswaDitemukan.nis, siswaDitemukan.nama, siswaDitemukan.kelas, tanggalStr, jamStr, statusKehadiran]); return ContentService.createTextOutput(JSON.stringify({ status: "success", message: siswaDitemukan.nama + " berhasil absen (" + statusKehadiran + ")", nama: siswaDitemukan.nama, kelas: siswaDitemukan.kelas, waktu: jamStr, keterangan: statusKehadiran })).setMimeType(ContentService.MimeType.JSON); } function tambahKelas(namaKelas) { return ContentService.createTextOutput(JSON.stringify({status: "success", message: "Kelas ditambahkan"})).setMimeType(ContentService.MimeType.JSON); } function tambahGuru(params) { return ContentService.createTextOutput(JSON.stringify({status: "success", message: "Guru ditambahkan"})).setMimeType(ContentService.MimeType.JSON); }
